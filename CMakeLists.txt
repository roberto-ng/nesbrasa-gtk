cmake_minimum_required(VERSION 3.28)

project(nesbrasa VERSION 0.1.0 LANGUAGES C CXX)

include(GNUInstallDirs)
include(CTest)
include(CMakePackageConfigHelpers)

set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_subdirectory(core)
option(BUILD_WEB "Build the Emscripten web frontend" OFF)
option(BUILD_GTK "Build the GTK frontend" ON)
option(BUILD_QT "Build the Qt 6 QtWidgets frontend" OFF)
if(BUILD_WEB)
    add_subdirectory(frontends/web)
endif()
if(BUILD_GTK AND NOT BUILD_WEB)
    add_subdirectory(frontends/gtk)
endif()
if(BUILD_QT AND NOT BUILD_WEB)
    add_subdirectory(frontends/qt)
endif()
